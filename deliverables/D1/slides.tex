% ---------------------------------------------------------------- algorithms
\begin{frame}[fragile]{Problem 3 --- Two genuinely different algorithms}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{A --- Euler integral} (adaptive Simpson)
    \begin{lstlisting}[style=mini,basicstyle=\ttfamily\tiny]
BETA_A(x, y, tol, maxDepth):
  if x<=0 or y<=0: error   # VAL-01
  f(t): return t^(x-1)*(1-t)^(y-1)
  lo <- EPS if x<1 else 0  # inset
  hi <- 1-EPS if y<1 else 1  # singular
  return ADAPT(f, lo, hi, tol,
            SIMPSON(f,lo,hi), 0, maxDepth)

ADAPT(f, a, b, tol, whole, d, dMax):
  m <- (a+b)/2
  L <- SIMPSON(f,a,m); R <- SIMPSON(f,m,b)
  if d>=dMax or |L+R-whole|<=15*tol:  # REL-03
    return L+R + (L+R-whole)/15  # Richardson
  return ADAPT(f,a,m,tol/2,L,d+1,dMax)
       + ADAPT(f,m,b,tol/2,R,d+1,dMax)
    \end{lstlisting}
    {\tiny $O(N)$ evals; intuitive ``area under curve''.}
    \column{0.5\textwidth}
    \textbf{B --- Gamma identity} (Lanczos $\ln\Gamma$, log domain)
    \begin{lstlisting}[style=mini,basicstyle=\ttfamily\tiny]
BETA_B(x, y):
  if x<=0 or y<=0: error   # VAL-01
  return EXP( LNGAMMA(x) + LNGAMMA(y)
              - LNGAMMA(x+y) )   # REL-02

LNGAMMA(z):        # g=7, coeffs P[0..8]
  if z < 0.5:                    # reflection
    return LN(pi) - LN(|SIN(pi*z)|)
           - LNGAMMA(1-z)
  z <- z - 1
  a <- P[0]
  for i in 1..8:                 # fixed loop
    a <- a + P[i]/(z+i)          #   -> REL-03
  t <- z + 7.5
  return 0.5*LN(2*pi)
       + (z+0.5)*LN(t) - t + LN(a)
    \end{lstlisting}
    {\tiny $O(1)$; log domain avoids $\Gamma$ overflow.}
  \end{columns}
  \vspace{-6pt}
  \begin{alertblock}{Verified difference (drives P4)}
    \tiny Both coded vs.\ \texttt{lgamma}: B matches $\sim$10 digits everywhere;
    \textbf{A fails ACC-01} near the singular corner --- $B(0.2,0.3)=7.7277$ vs.\ true
    $7.7485$ (rel.\ err $2.7\times10^{-3}$).
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Problem 4 --- CLI prototype (\texttt{src/cli.py})}

def beta(x, y):                       # Algorithm B, O(1)

  \end{lstlisting}

\begin{frame}{GAI prompt --- P1 Persona}
  \small \textbf{Type:} generative + decision-support.
  \begin{description}\footnotesize
    \item[TASK] Compare $\ge 3$ persona templates (goal-directed, role-based, lightweight)
      as a mind map and recommend one for a $B(x,y)$ calculator user; then draft one
      primary persona (name, role, skills, goals, education, tasks, environment,
      platform, Beta-specific needs/pains) $+$ a usage scenario; label evidence vs.\
      synthesis vs.\ assumption.
    \item[RESTRICTIONS] No invented statistics; not a real interviewed person; every
      attribute traceable to $\ge 5$ later requirements; no implementation tech; keep it
      specific, not boilerplate.
    \item[OUTPUT] Template mind-map outline $+$ decision, then a structured persona card
      $+$ assumptions list $+$ 3--5 sentence scenario (headings so the two parts split
      across slides).
  \end{description}
\end{frame}

\begin{frame}{GAI prompt --- P2 Requirements}
  \small \textbf{Type:} generative + transformational, with review. \emph{Informed by P1.}
  \begin{description}\footnotesize
    \item[TASK] Express requirements in ISO/IEC/IEEE 29148 style, \emph{informed by the
      pasted P1 persona}; one consistent statement style, category-grouped IDs; record
      id / statement / rationale (persona trace) / priority / verification; cover input,
      compute, display, retry, exit, and rejection of invalid inputs; then review against
      the 29148 quality characteristics.
    \item[RESTRICTIONS] No algorithm, library, or language inside a requirement; no vague
      terms unless quantified; each requirement singular and independently verifiable;
      only persona-justified.
    \item[OUTPUT] Requirements table (ID, statement, rationale, priority, verification)
      $+$ a separate assumptions list $+$ a review-findings table; LaTeX-reproducible.
  \end{description}
\end{frame}

\begin{frame}{GAI prompt --- P3 Algorithms}
  \small \textbf{Type:} generative + comparative. \emph{Informed by P2.}
  \begin{description}\footnotesize
    \item[TASK] Specify two genuinely different, language-neutral algorithms satisfying the
      pasted P2 IDs --- \textbf{A} adaptive-quadrature of the Euler integral (endpoint
      handling for $0<x,y<1$); \textbf{B} the Gamma identity via a Lanczos $\ln\Gamma$ in
      the log domain. For each: inputs, pre/postconditions, tolerance, validation,
      steps, stopping condition, max-work safeguard, complexity, weaknesses, worked
      trace; confirm they are substantially different.
    \item[RESTRICTIONS] No real-language syntax; every loop/recursion must terminate or
      have a max-work cap; no built-in Beta/Gamma; make trade-offs explicit.
    \item[OUTPUT] Two pseudocode listings, each $+$ pre/post, complexity/limits, trace;
      then a ``how A and B differ'' table.
  \end{description}
\end{frame}

\begin{frame}{GAI prompt --- P4 Selection + CLI}
  \small \textbf{Type:} decision-support + generative. \emph{Informed by P3.}
  \begin{description}\footnotesize
    \item[TASK] (1)~Selection: mind-map compare the two P3 algorithms over accuracy,
      domain coverage, stability, effort, D2 from-scratch fit, performance, and
      explainability; state the choice and why the other is rejected. (2)~A Python 3 CLI
      of the chosen algorithm: I/O separated from compute, prompts $+$ domain hint,
      validate, compute, formatted precision, helpful messages, sample runs.
    \item[RESTRICTIONS] Selection from the criteria, not preference; consistent with P2
      and anticipates the D2 from-scratch constraint; runs from a terminal; no unhandled
      traceback; compute core callable independently; flag math calls needing a D2
      replacement.
    \item[OUTPUT] Selection mind-map $+$ decision, full runnable code, and sample runs
      (valid / boundary / invalid) $+$ the run command.
  \end{description}
\end{frame}
